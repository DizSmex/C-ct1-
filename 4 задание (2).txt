using System;

public class Task4
{
    public int findFirst(int[] arr, int x)
    {
        for (int i = 0; i < arr.Length; i++)
            if (arr[i] == x) return i;
        return -1;
    }

    public int maxAbs(int[] arr)
    {
        int best = arr[0];
        for (int i = 1; i < arr.Length; i++)
            if (Math.Abs(arr[i]) > Math.Abs(best))
                best = arr[i];
        return best;
    }

    public int[] add(int[] arr, int[] ins, int pos)
    {
        int[] res = new int[arr.Length + ins.Length];

        for (int i = 0; i < pos; i++)
            res[i] = arr[i];

        for (int i = 0; i < ins.Length; i++)
            res[pos + i] = ins[i];

        for (int i = pos; i < arr.Length; i++)
            res[ins.Length + i] = arr[i];

        return res;
    }

    public int[] reverseBack(int[] arr)
    {
        int[] res = new int[arr.Length];
        for (int i = 0; i < arr.Length; i++)
            res[i] = arr[arr.Length - 1 - i];
        return res;
    }

    public int[] findAll(int[] arr, int x)
    {
        int count = 0;
        for (int i = 0; i < arr.Length; i++)
            if (arr[i] == x) count++;

        int[] res = new int[count];
        int k = 0;
        for (int i = 0; i < arr.Length; i++)
            if (arr[i] == x) res[k++] = i;

        return res;
    }


    private static int[] ReadArray(int size)
    {
        int[] arr = new int[size];
        for (int i = 0; i < size; i++)
        {
            Console.Write($"arr[{i}] = ");
            arr[i] = int.Parse(Console.ReadLine());
        }
        return arr;
    }

    public static void Main()
    {
        Console.OutputEncoding = System.Text.Encoding.UTF8;
        Task4 t = new Task4();

        Console.WriteLine("--- Задача 1. Поиск первого значения ---");
        Console.Write("Введите размер массива: ");
        int size1 = int.Parse(Console.ReadLine());
        int[] arr1 = ReadArray(size1);
        Console.Write("Введите число x: ");
        int x1 = int.Parse(Console.ReadLine());
        Console.WriteLine($"Результат: {t.findFirst(arr1, x1)}");

        Console.WriteLine("\n--- Задача 3. Поиск максимального по модулю ---");
        Console.Write("Введите размер массива: ");
        int size3 = int.Parse(Console.ReadLine());
        int[] arr3 = ReadArray(size3);
        Console.WriteLine($"Результат: {t.maxAbs(arr3)}");

        Console.WriteLine("\n--- Задача 5. Добавление массива в массив ---");
        Console.Write("Введите размер первого массива: ");
        int size5 = int.Parse(Console.ReadLine());
        int[] arr5 = ReadArray(size5);
        Console.Write("Введите размер второго массива: ");
        int size5b = int.Parse(Console.ReadLine());
        int[] ins5 = ReadArray(size5b);
        Console.Write("Введите позицию для вставки: ");
        int pos5 = int.Parse(Console.ReadLine());
        int[] res5 = t.add(arr5, ins5, pos5);
        Console.WriteLine($"Результат: [{string.Join(", ", res5)}]");

        Console.WriteLine("\n--- Задача 7. Возвратный реверс ---");
        Console.Write("Введите размер массива: ");
        int size7 = int.Parse(Console.ReadLine());
        int[] arr7 = ReadArray(size7);
        int[] res7 = t.reverseBack(arr7);
        Console.WriteLine($"Результат: [{string.Join(", ", res7)}]");

        Console.WriteLine("\n--- Задача 9. Все вхождения ---");
        Console.Write("Введите размер массива: ");
        int size9 = int.Parse(Console.ReadLine());
        int[] arr9 = ReadArray(size9);
        Console.Write("Введите число x: ");
        int x9 = int.Parse(Console.ReadLine());
        int[] res9 = t.findAll(arr9, x9);
        Console.WriteLine($"Результат: [{string.Join(", ", res9)}]");

        Console.ReadKey();
    }
}
