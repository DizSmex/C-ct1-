using System;

namespace ConsoleAppTask2
{
    public class Task2
    {
        public int abs(int x)
        {
            return x < 0 ? -x : x;
        }

        public bool is35(int x)
        {
            bool d3 = x % 3 == 0;
            bool d5 = x % 5 == 0;
            return (d3 || d5) && !(d3 && d5);
        }

        public int max3(int x, int y, int z)
        {
            int max = x;
            if (y > max) max = y;
            if (z > max) max = z;
            return max;
        }

        public int sum2(int x, int y)
        {
            int s = x + y;
            if (s >= 10 && s <= 19) return 20;
            return s;
        }

        public string day(int x)
        {
            switch (x)
            {
                case 1: return "понедельник";
                case 2: return "вторник";
                case 3: return "среда";
                case 4: return "четверг";
                case 5: return "пятница";
                case 6: return "суббота";
                case 7: return "воскресенье";
                default: return "это не день недели";
            }
        }

        public static void Main()
        {
            Console.OutputEncoding = System.Text.Encoding.UTF8;
            Task2 t = new Task2();

            Console.WriteLine("--- Задача 1. Модуль числа ---");
            Console.Write("Введите число x: ");
            int x1 = int.Parse(Console.ReadLine());
            Console.WriteLine($"Результат: {t.abs(x1)}");

            Console.WriteLine("\n--- Задача 3. Тридцать пять ---");
            Console.Write("Введите число x: ");
            int x3 = int.Parse(Console.ReadLine());
            Console.WriteLine($"Результат: {t.is35(x3)}");

            Console.WriteLine("\n--- Задача 5. Тройной максимум ---");
            Console.Write("Введите число x: ");
            int x5 = int.Parse(Console.ReadLine());
            Console.Write("Введите число y: ");
            int y5 = int.Parse(Console.ReadLine());
            Console.Write("Введите число z: ");
            int z5 = int.Parse(Console.ReadLine());
            Console.WriteLine($"Результат: {t.max3(x5, y5, z5)}");

            Console.WriteLine("\n--- Задача 7. Двойная сумма ---");
            Console.Write("Введите число x: ");
            int x7 = int.Parse(Console.ReadLine());
            Console.Write("Введите число y: ");
            int y7 = int.Parse(Console.ReadLine());
            Console.WriteLine($"Результат: {t.sum2(x7, y7)}");

            Console.WriteLine("\n--- Задача 9. День недели ---");
            Console.Write("Введите число от 1 до 7: ");
            int x9 = int.Parse(Console.ReadLine());
            Console.WriteLine($"Результат: {t.day(x9)}");

            Console.ReadKey();
        }
    }
}
